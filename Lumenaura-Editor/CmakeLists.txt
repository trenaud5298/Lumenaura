
file(GLOB_RECURSE LUMENAURA_EDITOR_HEADER_FILES "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h")
file(GLOB_RECURSE LUMENAURA_EDITOR_SRC_FILES "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cu")
message(STATUS "Cmake Current Source Director For Lumenaur-Editor: ${CMAKE_CURRENT_SOURCE_DIR}")

add_executable(Lumenaura_Editor)

target_link_libraries(Lumenaura_Editor
    PRIVATE Lumenaura_Core
)

target_include_directories(Lumenaura_Editor
    PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include"
)

target_sources(Lumenaura_Editor
    PRIVATE ${LUMENAURA_EDITOR_SRC_FILES}
)

source_group("Header Files" FILES ${LUMENAURA_EDITOR_HEADER_FILES})
source_group("Source Files" FILES ${LUMENAURA_EDITOR_SRC_FILES})
